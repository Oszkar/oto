```dart
import 'package:riverpod_annotation/riverpod_annotation.dart';

import '../rust/api.dart' as rust_api;
import 'discovery.dart';
import 'household.dart';
import 'model/household.dart';

part 'home_view_state.g.dart';

sealed class HomeViewState {
  const HomeViewState();
}

final class HomeInitialLoading extends HomeViewState {
  const HomeInitialLoading();
}

final class HomeDiscoveringWithCache extends HomeViewState {
  const HomeDiscoveringWithCache(this.household);
  final Household household;
}

final class HomeDiscoveryFailedNoCache extends HomeViewState {
  const HomeDiscoveryFailedNoCache(this.error);
  final Object error;
}

final class HomeDiscoveryFailedWithCache extends HomeViewState {
  const HomeDiscoveryFailedWithCache(this.household, this.error);
  final Household household;
  final Object error;
}

final class HomeEmpty extends HomeViewState {
  const HomeEmpty();
}

final class HomeReady extends HomeViewState {
  const HomeReady(this.household);
  final Household household;
}

@riverpod
HomeViewState homeViewState(Ref ref) {
  final discovery = ref.watch(discoveryProvider);
  final household = ref.watch(householdProvider);
  final hasCache = household.rooms.isNotEmpty || household.groups.isNotEmpty;

  if (discovery.isLoading) {
    return hasCache
        ? HomeDiscoveringWithCache(household)
        : const HomeInitialLoading();
  }

  if (discovery.hasError) {
    final error = discovery.error ?? StateError('Discovery failed');
    return hasCache
        ? HomeDiscoveryFailedWithCache(household, error)
        : HomeDiscoveryFailedNoCache(error);
  }

  final topology = discovery.valueOrNull;
  if (_isEmptyTopology(topology)) {
    return const HomeEmpty();
  }

  return HomeReady(household);
}

bool _isEmptyTopology(rust_api.Topology? topology) =>
    topology != null && topology.speakers.isEmpty && topology.groups.isEmpty;
```

```dart
  Widget build(BuildContext context, WidgetRef ref) {
```

```dart
    final rooms = household.rooms.values.toList()
      ..sort((a, b) => a.name.compareTo(b.name));

    return SettingsSection(
      title: 'Devices',
      children: [
        for (var i = 0; i < rooms.length; i++)
          DeviceRow(
            room: rooms[i],
            group: household.groups[rooms[i].groupId],
            roomsById: household.rooms,
            last: i == rooms.length - 1,
          ),
        if (rooms.isEmpty)
          const SettingsRow(

            label: 'No devices',
            subtitle: 'Scan the network from Home to discover speakers.',
            last: true,
          ),
      ],
    );
  }
}

class DeviceRow extends StatelessWidget {
  const DeviceRow({
    super.key,
    required this.room,
    required this.group,
    required this.roomsById,
    required this.last,
  });

  final RoomState room;
  final GroupState? group;
  final Map<String, RoomState> roomsById;
  final bool last;

  @override
  Widget build(BuildContext context) {
    final groupLabel = _groupLabel(group, roomsById);
    final status = room.online ? groupLabel : 'Offline';
    final icon = room.kind == RoomKind.soundbar ? 'soundbar' : 'speaker';
    return SettingsRow(
      icon: icon,
      label: room.name,
      subtitle: room.model ?? 'Speaker',
      last: last,
      trailing: _StatusPill(label: status, offline: !room.online),
    );
  }
}

String _groupLabel(GroupState? group, Map<String, RoomState> roomsById) {
  if (group == null || group.memberIds.length <= 1) return 'Standalone';
  final names = group.memberIds
      .map((id) => roomsById[id]?.name)
      .whereType<String>()
      .toList();
  return '${names.length} rooms';
}

class _StatusPill extends StatelessWidget {
  const _StatusPill({required this.label, required this.offline});

  final String label;
  final bool offline;

  @override
  Widget build(BuildContext context) {
    final oto = context.oto;
    return Container(
      padding: const EdgeInsets.symmetric(
        horizontal: Space.md8,
        vertical: Space.xs4,
      ),
      decoration: BoxDecoration(
        color: offline ? oto.danger.withOpacity(0.08) : oto.fillStrong,
        borderRadius: BorderRadius.circular(Radius_.control7),
      ),
      child: Text(
        label,
        style: TextStyles.label.copyWith(
          color: offline ? oto.danger : oto.inkMute,
          fontWeight: FontWeight.w700,
        ),
      ),
```
